//适合对行为进行记录、撤销或重做的场景

public interface Command {
    void execute();
    void undo();
}

//ConcreteCommand
public class LightOnCommand implements Command {
    //持有 receiver
    LightReceiver lightReceiver;
    @Override
    public void execute() {
        lightReceiver.on();
    }

    @Override
    public void undo() {
        lightReceiver.off();
    }
}

public class LightOffCommand implements Command {
    //持有 receiver
    LightReceiver lightReceiver;
    @Override
    public void execute() {
        lightReceiver.off();
    }

    @Override
    public void undo() {
        lightReceiver.on();
    }
}
public class NoCommand implements Command {
    @Override
    public void execute() {
    }
    @Override
    public void undo() {
    }
}

//请求的执行者 真正的命令执行对象  Receiver
public class LightReceiver {
    public void on(){
        System.out.println(" light on");
    }
    public void off(){
        System.out.println(" light off");
    }
}
