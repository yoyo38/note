1。 # 在你的用户目录下创建一个 nginx-docker 文件夹，结构清晰
mkdir -p ~/nginx-docker/{conf,conf.d,html,logs}

2。 
# 临时运行一个Nginx容器，并复制出默认的配置文件到宿主机目录
docker run --rm nginx:latest cat /etc/nginx/nginx.conf > ~/nginx-docker/conf/nginx.conf
docker run --rm nginx:latest cat /etc/nginx/conf.d/default.conf > ~/nginx-docker/conf.d/default.conf

3。 
docker run -d \
  --name production-nginx \
  -p 80:80 \
  -p 443:443 \
  -v ~/nginx-docker/conf/nginx.conf:/etc/nginx/nginx.conf:ro \
  -v ~/nginx-docker/conf.d:/etc/nginx/conf.d:ro \
  -v ~/nginx-docker/html:/usr/share/nginx/html:ro \
  -v ~/nginx-docker/logs:/var/log/nginx \
  --restart unless-stopped \
  nginx:latest

4。 测试
echo '<h1>Hello from Docker Nginx!</h1>' > ~/nginx-docker/html/index.html

5。 docker compose
# docker-compose.yml
version: '3.8'

services:
  nginx:
    image: nginx:latest
    container_name: production-nginx
    ports:
      - "80:80"
      - "443:443"
    volumes:
      - ./conf/nginx.conf:/etc/nginx/nginx.conf:ro
      - ./conf.d:/etc/nginx/conf.d:ro
      - ./html:/usr/share/nginx/html:ro
      - ./logs:/var/log/nginx
    restart: unless-stopped



# 启动服务（-d 表示后台运行）
docker compose up -d

# 停止服务
docker compose down

# 查看日志
docker compose logs -f



# 允许80端口
sudo ufw allow 80/tcp
# 允许443端口
sudo ufw allow 443/tcp
# 重新加载防火墙
sudo ufw reload

chmod -R 755 ~/nginx-docker
